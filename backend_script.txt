const MAIN_SS_ID = "15_vgd-bZPgcEj-l2r5uYGXXujKTyk-1XOGJ8eDAbhRw";
const PARENT_FOLDER_ID = "1giAjJCqeALcAHy5RiG8WPeCpr0mAFqT8";

function doGet(e) {
  const action = e.parameter.action;
  if (action === 'sync') {
    return ContentService.createTextOutput(JSON.stringify(getAllData()))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput("EC VAN HỮU GIA Backend is running...");
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.openById(MAIN_SS_ID);

    // --- TRƯỜNG HỢP 1: CẬP NHẬT KIỂM KÊ (UPDATE ASSET) ---
    if (data.action === "updateAsset") {
      const sheet = ss.getSheetByName("Tài sản theo xe");
      const rows = sheet.getDataRange().getValues();
      const headers = rows[0];
      
      // Tìm cột Biển số xe và cột vật tư cần sửa
      const plateCol = headers.findIndex(h => h.toLowerCase().includes("biển số"));
      const assetCol = headers.findIndex(h => h.toLowerCase() === data.assetName.toLowerCase());
      
      if (plateCol === -1 || assetCol === -1) throw new Error("Không tìm thấy cột dữ liệu");

      // Tìm dòng chứa biển số xe
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][plateCol]).replace(/[^a-zA-Z0-9]/g, '').toLowerCase() === 
            data.licensePlate.replace(/[^a-zA-Z0-9]/g, '').toLowerCase()) {
          sheet.getRange(i + 1, assetCol + 1).setValue(data.newValue);
          return ContentService.createTextOutput(JSON.stringify({ success: true }))
            .setMimeType(ContentService.MimeType.JSON);
        }
      }
      throw new Error("Không tìm thấy biển số xe");
    }

    // --- TRƯỜNG HỢP 2: BÀN GIAO XE (HANDOVER) ---
    const sheet = ss.getSheetByName("Lịch sử bàn giao") || ss.insertSheet("Lịch sử bàn giao");
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Biển số xe", "Số KM (ODO)", "Người giao", "Người nhận", "Ghi chú", "Ngày bàn giao", "Link ảnh biên bản", "Thời gian hệ thống"]);
    }

    let imageUrl = "";
    if (data.image) {
      const rootFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);
      
      // 1. Tạo thư mục Ngày (Ví dụ: 10-05-2024)
      const dateStr = data.actualDate.replace(/\//g, "-");
      const dateFolder = getOrCreateFolder(rootFolder, dateStr);
      
      // 2. Tạo thư mục Biển số xe bên trong thư mục Ngày
      const plateFolder = getOrCreateFolder(dateFolder, data.licensePlate);
      
      const fileName = "IMG_" + data.licensePlate + "_" + new Date().getTime() + ".jpg";
      const blob = Utilities.newBlob(Utilities.base64Decode(data.image.split(",")[1]), "image/jpeg", fileName);
      const file = plateFolder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      imageUrl = file.getUrl();
    }

    sheet.appendRow([
      data.licensePlate,
      data.mileage,
      data.sender,
      data.receiver,
      data.notes,
      data.actualDate,
      imageUrl,
      new Date()
    ]);

    return ContentService.createTextOutput(JSON.stringify({ success: true, url: imageUrl }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Hàm hỗ trợ tìm hoặc tạo thư mục
function getOrCreateFolder(parent, name) {
  const folders = parent.getFoldersByName(name);
  if (folders.hasNext()) return folders.next();
  return parent.createFolder(name);
}

function getAllData() {
  const mainSS = SpreadsheetApp.openById(MAIN_SS_ID);

  return {
    vehicles: getSheetData(mainSS, "Danh sách xe"),
    drivers: getSheetData(mainSS, "Danh sách tài xế"),
    assets: getSheetData(mainSS, "Tài sản theo xe"),
    handoverLogs: getSheetData(mainSS, "Lịch sử bàn giao")
  };
}

function getSheetData(ss, sheetName) {
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length === 0) return [];
  const headers = data.shift();
  return data.map(row => {
    let obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
}
